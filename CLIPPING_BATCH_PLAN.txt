- [ ] Task 2.2: Implement Validated NVENC Render Function

  ```javascript
  const { execFileSync } = require('child_process');
  const path = require('path');
  const ffmpegPath = require('ffmpeg-static');

  function renderSplitScreenHardware(arollPath, brollPath, assPath, outputPath, cropX = null) {
    const escapedAss = assPath.replace(/\\/g, '/').replace(/:/g, '\\:');
    const arollCrop = cropX !== null
      ? `crop=1080:960:${cropX}:0`
      : 'scale=1080:960:force_original_aspect_ratio=increase,crop=1080:960';

    const filterComplex = [
      `[0:v]${arollCrop},setsar=1[top]`,
      `[1:v]scale=1080:960:force_original_aspect_ratio=increase,crop=1080:960,setsar=1[bot]`,
      `[top][bot]vstack=inputs=2[stacked]`,
      `[stacked]ass='${escapedAss}'[outv]`
    ].join(';');

    const args = [
      '-y',
      '-i', arollPath,
      '-stream_loop', '-1', '-i', brollPath,
      '-filter_complex', filterComplex,
      '-map', '[outv]',
      '-map', '0:a?',
      '-c:v', 'h264_nvenc',
      '-preset', 'p4',
      '-tune', 'hq',
      '-rc:v', 'vbr',
      '-cq:v', '20',
      '-b:v', '0',
      '-pix_fmt', 'yuv420p',
      '-c:a', 'aac',
      '-b:a', '192k',
      '-shortest',
      outputPath
    ];

    execFileSync(ffmpegPath, args, { stdio: 'inherit' });
  }
  ```
